# Feed Page (Slice #2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the placeholder feed at `/` with the real inform-and-direct hub: a "Needs your attention" actionable band above a capped chronological activity stream, fed by a single capture point in `BrowserDao.executeRemote` plus CRABS' native change-callback for push-based reactivity.

**Architecture:** Pure data-projection code lives in `shared/src/feed.ts` (jest-covered, TDD'd): the `FeedEvent` type plus a pure op→event builder. Browser-side, `BrowserDao` captures consequential ops during `executeRemote` (after the existing mirrors) and subscribes CRABS' `node.on('change')` via a ~50-line vendored-wrapper resync; the dao store forwards notifications to its `bump()` so all pages update instantly. Four presentation components compose the page; auth lands on `/`.

**Tech Stack:** Vue 3 SFC + pinia (from slice #1), shared TS modules (jest + ts-jest), CRABS wasm bindings + `bindings/devtools` trampoline pattern.

```ts
import { DIMENSIONS, HarmVerdict } from './contribution';

export type FeedEventType =
  | 'proposal_created' | 'proposal_executed'
  | 'contribution_submitted' | 'contribution_verified'
  | 'contribution_settled' | 'contribution_appealed'
  | 'election_started' | 'election_finalized'
  | 'round_audited' | 'round_reckoned' | 'round_published';

export interface FeedEvent {
  index: number;
  type: FeedEventType;
  actor: string;
  at: number;
  round: number;
  summary: string;
  targetId?: string;
  detail?: {
    proposalType?: 'direct' | 'quadratic';
    verifyPass?: boolean;
    harm?: HarmVerdict;
    fair?: boolean;
    aggregatedCount?: number;
  };
}

// The mirrored-timeline op set — votes are deliberately absent (D2).
export const CONSEQUENTIAL_OPS: ReadonlySet<string> = new Set([
  'create_proposal', 'execute',
  'submit_contribution', 'verify_contribution', 'settle_contribution', 'appeal_verdict',
  'start_election', 'finalize_election',
  'audit_round', 'reckon_round', 'complete_round',
]);

const OP_TO_EVENT: Record<string, FeedEventType> = {
  create_proposal: 'proposal_created',
  execute: 'proposal_executed',
  submit_contribution: 'contribution_submitted',
  verify_contribution: 'contribution_verified',
  settle_contribution: 'contribution_settled',
  appeal_verdict: 'contribution_appealed',
  start_election: 'election_started',
  finalize_election: 'election_finalized',
  audit_round: 'round_audited',
  reckon_round: 'round_reckoned',
  complete_round: 'round_published',
};

function dimLabels(dims: Record<string, number>): string[] {
  return Object.keys(dims).map((k) => {
    const def = DIMENSIONS[Number(k)];
    return def ? `${def.cls} (${def.code})` : 'C_?';
  });
}

// Pure projection: one signed op → one FeedEvent (or null for ops the feed
// does not mirror). `currentRound` is round:current AT capture; complete_round
// bumps that register before this runs, so its event belongs to the finishing
// round (same target logic as the dao's round mirror).
export function feedEventFromOp(
  type: string,
  signerId: string,
  payloadJson: string,
  at: number,
  currentRound: number,
  nextIndex: number,
): FeedEvent | null {
  const eventType = OP_TO_EVENT[type];
  if (!eventType) return null;

  let payload: any = {};
  try { payload = JSON.parse(payloadJson.replace(/\0$/, '')); } catch { payload = null; }

  const round = type === 'complete_round' ? currentRound - 1 : currentRound;
  const base: FeedEvent = { index: nextIndex, type: eventType, actor: signerId, at, round };

  if (payload === null) {
    // Corrupt/unknown payload on a consequential op: honest generic line.
    return { ...base, summary: `${type.replace(/_/g, ' ')}` };
  }

  switch (type) {
    case 'create_proposal': {
      const proposalType = payload.proposalType === 'quadratic' ? 'quadratic' : 'direct';
      const title = typeof payload.title === 'string' ? payload.title : '';
      return {
        ...base,
        summary: `"${title}" — ${proposalType} proposal created`,
        targetId: typeof payload.proposalId === 'string' ? payload.proposalId : undefined,
        detail: { proposalType },
      };
    }
    case 'execute': {
      const id = typeof payload.proposalId === 'string' ? payload.proposalId : '';
      return { ...base, summary: `proposal ${id.slice(0, 8)} executed`, targetId: id || undefined };
    }
    case 'submit_contribution': {
      const dims = dimLabels(payload.dims ?? {}).join(', ');
      const summary = typeof payload.summary === 'string' ? payload.summary : '';
      return {
        ...base,
        summary: `${dims || 'C_?'} contribution submitted — "${summary}"`,
        targetId: typeof payload.contributionId === 'string' ? payload.contributionId : undefined,
      };
    }
    case 'verify_contribution': {
      const submitter = typeof payload.submitter === 'string' ? payload.submitter : '';
      const pass = payload.pass === true;
      const harm = (payload.harm ?? 'none') as HarmVerdict;
      const harmNote = harm !== 'none' && pass ? ` (${harm})` : '';
      return {
        ...base,
        summary: `${signerId} verified ${submitter} — ${pass ? 'upheld' : 'no'}${harmNote}`,
        targetId: typeof payload.contributionId === 'string' ? payload.contributionId : undefined,
        detail: { verifyPass: pass, harm },
      };
    }
    case 'settle_contribution':
      return { ...base, summary: `${payload.submitter ?? ''} settled`.trim(),
        targetId: typeof payload.contributionId === 'string' ? payload.contributionId : undefined };
    case 'appeal_verdict':
      return { ...base, summary: `${signerId} appealed — awaiting re-verification`,
        targetId: typeof payload.contributionId === 'string' ? payload.contributionId : undefined };
    case 'start_election':
      return {
        ...base,
        summary: `custodian election started — ${Array.isArray(payload.candidates) ? payload.candidates.length : 0} candidates`,
        targetId: typeof payload.electionId === 'string' ? payload.electionId : undefined,
      };
    case 'finalize_election':
      return { ...base, summary: 'custodian election finalized',
        targetId: typeof payload.electionId === 'string' ? payload.electionId : undefined };
    case 'audit_round': {
      const fair = payload.fair === true;
      return { ...base, summary: `round ${round} audited — ${fair ? 'fair' : 'closed with debt'}`,
        detail: { fair } };
    }
    case 'reckon_round':
      return { ...base, summary: `round ${round} reckoned — ready to complete` };
    case 'complete_round': {
      const count = Array.isArray(payload.entries) ? payload.entries.length : 0;
      return { ...base, summary: `round ${round} published — ${count} contribution(s) aggregated`,
        detail: { aggregatedCount: count } };
    }
    default:
      return null;
  }
}
```

```ts
// ============================================================
// Change events
// ============================================================

// Module-wide change-event plumbing: one trampoline per WASM module,
// dispatching to every Node's listeners (filtered by node id, which is the
// bootstrap admin id stamped into op->node_id).
const _changeListenerRegistry = new WeakMap();

function _getChangeListeners(M) {
  if (!_changeListenerRegistry.has(M)) {
    _changeListenerRegistry.set(M, []);
  }
  const listeners = _changeListenerRegistry.get(M);
  if (!listeners._pending) listeners._pending = [];
  return listeners;
}

// Deliver one event to every matching listener. Listener failures are
// contained so one bad listener cannot starve the others.
function _dispatchChangeEvent(listeners, event) {
  for (const entry of listeners.slice()) {
    if (entry.node != null && event.node !== entry.node) continue;
    try {
      entry.listener(event);
    } catch (listenerError) {
      console.warn('crabs: change listener failed', listenerError);
    }
  }
}

function _ensureChangeTrampoline(M) {
  const listeners = _getChangeListeners(M);
  if (listeners._trampoline) return;
  const trampoline = M.addFunction((jsonPointer) => {
    const text = jsonPointer ? M.UTF8ToString(jsonPointer) : '';
    let event;
    try {
      event = JSON.parse(text);
    } catch (parseError) {
      return;  // malformed event: drop, never throw into C
    }
    if (listeners._dispatching) {
      listeners._pending.push(event);   // nested: defer until outer dispatch drains
      return;
    }
    listeners._dispatching = true;
    try {
      _dispatchChangeEvent(listeners, event);
      while (listeners._pending.length > 0) {
        _dispatchChangeEvent(listeners, listeners._pending.shift());
      }
    } finally {
      listeners._dispatching = false;
    }
  }, 'vi');
  listeners._trampoline = trampoline;
  M._crabs_wasm_set_change_trampoline(trampoline);
}
```

- [ ] **Step 2: add the methods to `Node`** — insert AFTER the constructor (after the `this._adminId = adminId;` block) as instance methods:

```ts
  // on('change', listener) returns an unsubscribe function (upstream parity).
  // Trust note: change events are observational — listeners must not mutate
  // node state; nested mutations inside listeners are deferred upstream.
  on(eventName, listener) {
    if (eventName !== 'change' || typeof listener !== 'function') return;
    _ensureChangeTrampoline(this._M);
    const listeners = _getChangeListeners(this._M);
    listeners.push({ node: this._adminId, listener });
    const node = this;
    return () => {
      const entryIndex = listeners.findIndex(
        (entry) => entry.listener === listener && entry.node === node._adminId);
      if (entryIndex >= 0) listeners.splice(entryIndex, 1);
    };
  }

  off(eventName, listener) {
    if (eventName !== 'change') return;
    const listeners = _getChangeListeners(this._M);
    const entryIndex = listeners.findIndex(
      (entry) => entry.listener === listener && entry.node === this._adminId);
    if (entryIndex >= 0) listeners.splice(entryIndex, 1);
  }
```

```ts
  // Feed timeline capture: consequential ops only, newest first, bounded.
  // Runs AFTER execute() so handler effects (status registers) are already
  // applied — timestamps and round:current are exact at capture time.
  private async mirrorFeedEvent(op: any): Promise<void> {
    let payloadJson = '';
    try {
      const raw = op.payload;
      payloadJson = typeof raw === 'string' ? raw : new TextDecoder().decode(raw as Uint8Array);
    } catch {
      payloadJson = '';
    }
    const event = feedEventFromOp(
      op.type, op.signerId, payloadJson,
      this.getNodeTimeMs(), this.getCurrentRound(), ++this.feedIndex,
    );
    if (!event) return;
    this.feedEvents.unshift(event);
    if (this.feedEvents.length > 200) this.feedEvents.length = 200;
  }

  getFeedEvents(): FeedEvent[] {
    return Array.from(this.feedEvents);
  }
```

```vue
<template>
  <section id="needs-attention" aria-label="Needs your attention">
    <span class="eyebrow">Needs your attention</span>
    <div v-if="items.length === 0" class="empty-state caught-up">You're all caught up.</div>

      <RouterLink
        v-for="item in items"
        :key="item.key"
        :to="item.to"
        class="attention__card"
      >
        <span class="attention__label mono">{{ item.kind }}</span>
        <span class="attention__text">{{ item.text }}</span>
        <span class="attention__arrow">→</span>
      </RouterLink>
    </div>
  </section>

<script setup lang="ts">
import { computed } from 'vue';
import { RouterLink } from 'vue-router';
import { wizardModel, type ContributionFacts } from '@shared/wizard';
import { schemaForDims } from '@shared/contribution';
import { STAGE_AUDITED, STAGE_OPEN } from '@shared/round';
import { useDaoStore } from '../../stores/dao';

const ds = useDaoStore();

type Item = { key: string; kind: string; text: string; to: string };

const items = computed<Item[]>(() => {
  void ds.tick;
  const d = ds.dao;
  const me = ds.username;
  if (!d || !me) return [];
  const now = d.getNodeTimeMs();
  const out: Item[] = [];

  for (const p of d.getProposals()) {
    const pType = d.getProposalType(p.proposalId);
    if (d.isProposalExecuted(p.proposalId)) continue;
    if (Math.max(0, d.getProposalExpiry(p.proposalId) - now) <= 0) continue;
    const voted = d.getProposalVoteCount(p.proposalId, me) > 0;
    const nextCost = ((d.getProposalVoteCount(p.proposalId, me) + 1) * (d.getProposalVoteCount(p.proposalId, me) + 2) * (2 * d.getProposalVoteCount(p.proposalId, me) + 3)) / 6;
    const canAgain = pType === 'quadratic' && d.getVoteBalance(p.proposalId, me) >= nextCost && nextCost > 0;
    if (voted && !canAgain) continue;
    out.push({ key: `p-${p.proposalId}`, kind: 'proposal', text: `"${p.title}" is open for voting`, to: '/proposals' });
  }

  for (const entry of d.getContributions()) {
    if (entry.submitter === me) continue;
    const status = d.getContributionStatus(entry.record.contributionId);
    if (status === 'unknown' || status === 'accepted' || status === 'rejected') continue;
    const schema = schemaForDims(entry.record.dims);
    const facts: ContributionFacts = {
      status: 0,
      stepIndex: d.getContributionStepIndex(entry.record.contributionId),
      done: Object.fromEntries(schema.steps.map((s) => [
        s.stepId, d.getContributionStepDone(entry.record.contributionId, s.stepId),
      ])),
      submitter: entry.submitter,
    };
    const model = wizardModel(schema, facts, me, ds.custodians.includes(me));
    const step = model.steps[model.currentStepIndex];
    if (!step || !step.actors.some((actor) => actor === 'verifier' || actor === 'custodian' || actor === 'member')) continue;
    const roleInfo = step.roles.verifier ?? step.roles.custodian ?? step.roles.member;
    if (!roleInfo || !roleInfo.mayAct) continue;
    out.push({ key: `c-${entry.record.contributionId}`, kind: 'verify', text: `"${entry.record.summary}" awaits your verification`, to: '/contribute' });
  }

  for (const e of d.getElections()) {
    if (d.getElectionFinalized(e.electionId) !== 0) continue;
    if (Math.max(0, d.getElectionExpiry(e.electionId) - now) <= 0) continue;
    if (d.hasBallot(e.electionId)) continue;
    out.push({ key: `e-${e.electionId}`, kind: 'election', text: 'A custodian election is open for ballots', to: '/elections' });
  }

  const stage = d.getRoundStage(d.getCurrentRound());
  if (stage === STAGE_OPEN || stage === STAGE_AUDITED) {
    out.push({ key: 'round', kind: 'round', text: 'The round spine is waitinng for its next step', to: '/round' });
  }

  return out;
});
</script>
```

```js
  if (!(await page.url()).endsWith('/')) fail('post-login landing is not the feed');
```
